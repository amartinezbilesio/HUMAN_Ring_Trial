SIRIUS annotation - all MS2 per lab (strategy 3)

Goal

Annotate all MS2-bearing features from every lab, without requiring cross-lab consensus, and emit the two canonical CSVs defined in 5_downstream_analysis/CANONICAL_SCHEMAS.md:

  • annotation_identity.csv — one row per (mixture, lab, compound, polarity) with best ranks, adducts, COSMIC scores, and a confidence_tier placeholder ("unfiltered"; P5 overwrites).
  • annotation_chrom_metrics.csv — one row per (mixture, lab, polarity, compound, adduct, source) with areaUnderTic, xicFwhm, maxIntensity, apex_rt from the two-pass EIC pipeline.

This file is a pure producer: no cross-lab comparison plots, no abundance / RT / presence analysis. Those live in 5_downstream_analysis/9_per_compound_diagnostics.qmd.

1. Extraction helper

extract_all_ms2() loads the raw mzML files for one (lab, mixture) and returns the MS1+MS2 spectra ready for SIRIUS, split by polarity.

Acquisition layouts. afekta, cembio and icl have dedicated _MS2 files; HMGU acquires MS1+MS2 together in the regular sample file. For DDA labs (afekta, hmgu, icl) features are formed by fragmentGroupIndex() (one MS1 + its triggered MS2). Cembio acquires MS1 and MS2 in temporally disjoint blocks within one file with a targeted inclusion list, so co-elution-based pairing is impossible. Cembio features are formed by RT-proximity grouping: two MS2 of the same precursor m/z belong to the same feature iff their RT gap is ≤ cembio_block_gap_s (default 30 s). This replaces a legacy rle(precursorMz) approach that started a new feature every time another inclusion-list precursor fired between two MS2 of the same precursor — shattering one chromatographic peak into many features (e.g. ritalinic acid in mix 2.9 pos: one inclusion-list scan window at m/z 220.1335 fired 11 MS2 at 5-9 s intervals interleaved with other precursors → legacy rle made 11 features; proximity grouping correctly makes 1).

Noise filter. After loading we apply a per-MS2 intensity filter: keep peaks with intensity ≥ 5% of the base peak OR ≥ 1000 counts. This cuts MS2 peak counts ~4× on the noisy late-RT hmgu batches (median 91 → 21 peaks/MS2) while leaving normal spectra essentially untouched. The OR floor rescues mid-intensity fragments when the base peak is unusually tall.

extract_all_ms2 <- function(lab_name, study_group, mixture) {
  study_path <- here("1_preprocessing", lab_name, study_group)
  mzml_dir   <- file.path(study_path, "mzml")
  mix_label  <- mix_us(mixture)

  empty_result <- function(sp = Spectra(backend = MsBackendMemory())) {
    list(spectra_pos = sp, spectra_neg = sp,
         n_ms2_pos = 0L, n_ms2_neg = 0L,
         n_features_pos = 0L, n_features_neg = 0L)
  }

  seq_pos <- read_xlsx(file.path(study_path, "seq_pos.xlsx")) |> as.data.frame()
  seq_neg <- read_xlsx(file.path(study_path, "seq_neg.xlsx")) |> as.data.frame()
  seq_pos$polarity <- "pos"; seq_neg$polarity <- "neg"
  seq <- rbind(seq_pos, seq_neg)
  seq$filename <- paste0(seq$`Data File`, ".mzML")
  seq <- seq[seq$filename %in% list.files(mzml_dir), ]
  # Mixture is the last underscore-delimited token of the file basename,
  # after stripping a trailing `_MS2` and any `target` suffix (used by
  # cembio for targeted-MS2 acquisitions, e.g. "..._2.1target.mzML").
  seq$mixture <- gsub("\\.", "_",
                      sub("target$", "",
                          sub(".*_", "",
                              sub("_MS2$", "", seq$`Data File`))))

  has_ms2_files <- any(grepl("MS2", seq$filename, fixed = TRUE))
  seq <- if (has_ms2_files) {
    seq[grepl("MS2", seq$filename, fixed = TRUE) & seq$mixture == mix_label, ]
  } else {
    seq[seq$mixture == mix_label, ]
  }
  if (nrow(seq) == 0L) return(empty_result())

  sp <- Spectra(file.path(mzml_dir, seq$filename), backend = MsBackendMzR())
  sp <- setBackend(sp, MsBackendMemory())
  sp <- applyProcessing(sp)

  sp <- filterIntensity(sp, intensity = function(x) {
    if (length(x) == 0L) return(logical(0))
    bp <- max(x, na.rm = TRUE)
    if (!is.finite(bp) || bp <= 0) return(rep(TRUE, length(x)))
    x >= bp * 0.05 | x >= 1000
  }, msLevel. = 2L)

  if (lab_name == "cembio") {
    sp <- sp[sp$msLevel == 2L]
    sp <- filterEmptySpectra(sp)
    if (length(sp) == 0L) return(empty_result(sp))

    # Proximity-aware grouping for cembio's block-MS2 acquisition.
    #
    # Legacy `rle(precursorMz)` started a new feature every time another
    # inclusion-list precursor fired in between two MS2 of the same
    # precursor, shattering one chromatographic peak into many "features".
    # E.g. cembio mix 2.9 ritalinic acid pos: one inclusion-list scan
    # window at m/z 220.1335 produced 11 MS2 firings 5-9 s apart between
    # MS2 of other precursors → 11 separate group_idx values.
    #
    # New rule: sort by (file, precursorMz, RT). Two MS2 of the same
    # precursor m/z in the same file are in the same group iff their
    # RT gap is <= `cembio_block_gap_s`. This correctly collapses
    # interleaved firings of the same precursor while still splitting
    # truly distinct chromatographic peaks of the same m/z at well-
    # separated RTs.
    cembio_block_gap_s <- 30

    key       <- paste0(sp$dataOrigin, "|", sprintf("%.4f", sp$precursorMz))
    ord       <- order(key, sp$rtime)
    key_s     <- key[ord]
    rt_s      <- sp$rtime[ord]

    starts_new_group <- c(TRUE,
                          key_s[-1L] != key_s[-length(key_s)] |
                          (rt_s[-1L] - rt_s[-length(rt_s)]) > cembio_block_gap_s)
    grp_sorted <- cumsum(starts_new_group)

    grp <- integer(length(sp))
    grp[ord] <- grp_sorted
    sp$group_idx <- grp
  } else {
    keep <- unlist(lapply(split(seq_along(sp), sp$dataOrigin), function(idx) {
      first_ms1 <- which(sp$msLevel[idx] == 1L)[1L]
      if (is.na(first_ms1)) integer(0) else idx[first_ms1:length(idx)]
    }))
    sp <- sp[sort(keep)]
    sp <- filterEmptySpectra(sp)
    if (length(sp) == 0L) return(empty_result(sp))
    sp$group_idx <- fragmentGroupIndex(sp)
    sp <- sp[duplicated(sp$group_idx) | duplicated(sp$group_idx, fromLast = TRUE)]
    if (length(sp) == 0L) return(empty_result(sp))
  }

  list(
    spectra_pos    = sp[sp$polarity == 1L],
    spectra_neg    = sp[sp$polarity == 0L],
    n_ms2_pos      = sum(sp$msLevel == 2L & sp$polarity == 1L),
    n_ms2_neg      = sum(sp$msLevel == 2L & sp$polarity == 0L),
    n_features_pos = length(unique(sp$group_idx[sp$polarity == 1L])),
    n_features_neg = length(unique(sp$group_idx[sp$polarity == 0L]))
  )
}

run_sirius_for_lab()

Runs SIRIUS over one (lab, mixture)’s spectra in batches of batch_size features, caching per-batch results to CSV so re-renders skip completed batches. The function manages the SIRIUS server lifecycle internally and restarts the JVM every restart_every live batches - a long-running JVM accumulates state that eventually crashes import(), so bouncing the process clears it (the on-disk project survives the restart).

Each batch also cross-checks featuresId(srs, type = "xcms") against the features intended for import and re-imports anything silently dropped (up to two retries; bail when retries stop making progress).

Pos and neg features go into the same project per batch so RuSirius assigns polarity correctly. numberOfCandidates = 10/3 (per ionization) is a deliberate cut from the default 15/5 - each candidate triggers a CSI:FingerID + BIO lookup, the dominant per-feature cost.

run_sirius_for_lab <- function(sp_pos, sp_neg,
                               username, password, projectId, path,
                               batch_size    = 200,
                               cache_dir     = NULL,
                               cache_prefix  = NULL,
                               restart_every = 7L) {
  if (length(sp_pos) == 0L && length(sp_neg) == 0L)
    return(list(summary = NULL, full = NULL))

  start_sirius <- function() Sirius(username = username, password = password,
                                    projectId = projectId, path = path)
  srs <- start_sirius()
  on.exit(try(shutdown(srs), silent = TRUE), add = TRUE)

  groups_pos <- unique(sp_pos$group_idx)
  groups_neg <- unique(sp_neg$group_idx)
  pol_lookup <- c(
    setNames(rep("pos", length(groups_pos)), as.character(groups_pos)),
    setNames(rep("neg", length(groups_neg)), as.character(groups_neg))
  )

  all_groups <- c(groups_pos, groups_neg)
  batches    <- split(all_groups, ceiling(seq_along(all_groups) / batch_size))
  has_cache  <- !is.null(cache_dir) && !is.null(cache_prefix)

  sums         <- vector("list", length(batches))
  fulls        <- vector("list", length(batches))
  live_batches <- 0L

  for (i in seq_along(batches)) {
    batch_sum_path  <- if (has_cache) file.path(cache_dir, sprintf("%s_batch%d_sum.csv",  cache_prefix, i))
    batch_full_path <- if (has_cache) file.path(cache_dir, sprintf("%s_batch%d_full.csv", cache_prefix, i))

    message("  batch ", i, "/", length(batches))

    if (has_cache && file.exists(batch_sum_path) && file.exists(batch_full_path)) {
      message("    → cached, skipping")
      sums[[i]]  <- read.csv(batch_sum_path, stringsAsFactors = FALSE) |>
        mutate(across(any_of(c("confidenceExactMatch", "confidenceApproxMatch")), as.numeric),
               across(any_of(sirius_id_cols), as.character))
      fulls[[i]] <- read.csv(batch_full_path, stringsAsFactors = FALSE) |>
        mutate(across(any_of(sirius_id_cols), as.character))
      next
    }

    if (live_batches > 0L && live_batches %% restart_every == 0L) {
      message("    → restarting SIRIUS after ", live_batches, " live batches")
      try(shutdown(srs), silent = TRUE)
      srs <- start_sirius()
    }

    b_pos <- sp_pos[sp_pos$group_idx %in% batches[[i]]]
    b_neg <- sp_neg[sp_neg$group_idx %in% batches[[i]]]
    n_pos <- length(unique(b_pos$group_idx))
    n_neg <- length(unique(b_neg$group_idx))
    message("    ", n_pos, " pos + ", n_neg, " neg features")

    if (n_pos > 0)
      srs <- import(srs, spectra = b_pos, ms_column_name = "group_idx",
                    deleteExistingFeatures = TRUE)
    if (n_neg > 0)
      srs <- import(srs, spectra = b_neg, ms_column_name = "group_idx",
                    deleteExistingFeatures = n_pos == 0)

    expected_ids <- as.character(c(unique(b_pos$group_idx),
                                   unique(b_neg$group_idx)))
    project_ids  <- as.character(featuresId(srs, type = "xcms"))
    missing_ids  <- setdiff(expected_ids, project_ids)
    for (retry in 1:2) {
      if (length(missing_ids) == 0L) break
      message("    !! ", length(missing_ids), " missing - retry ", retry)
      miss_pos <- b_pos[b_pos$group_idx %in% missing_ids]
      miss_neg <- b_neg[b_neg$group_idx %in% missing_ids]
      if (length(miss_pos) > 0L)
        srs <- import(srs, spectra = miss_pos, ms_column_name = "group_idx",
                      deleteExistingFeatures = FALSE)
      if (length(miss_neg) > 0L)
        srs <- import(srs, spectra = miss_neg, ms_column_name = "group_idx",
                      deleteExistingFeatures = FALSE)
      prev_n      <- length(missing_ids)
      missing_ids <- setdiff(expected_ids,
                             as.character(featuresId(srs, type = "xcms")))
      if (length(missing_ids) >= prev_n) {
        message("    !! retry made no progress - giving up")
        break
      }
    }
    if (length(missing_ids) > 0L)
      warning("    !! ", length(missing_ids), " feature(s) never imported: ",
              paste(head(missing_ids, 10), collapse = ","))

    run(srs,
      formulaIdParams = formulaIdParam(
        numberOfCandidates              = 10,
        instrument                      = "QTOF",
        numberOfCandidatesPerIonization = 3,
        massAccuracyMS2ppm              = 10,
        filterByIsotopePattern          = FALSE,
        isotopeMs2Settings              = "SCORE",
        performDeNovoBelowMz            = 600),
      predictParams           = predictParam(),
      structureDbSearchParams = structureDbSearchParam(structureSearchDbs = c("BIO")),
      recompute = TRUE, wait = TRUE)

    sums[[i]] <- summary(srs, result.type = "structure") |>
      mutate(across(any_of(c("confidenceExactMatch", "confidenceApproxMatch")), as.numeric),
             across(any_of(sirius_id_cols), as.character),
             polarity = pol_lookup[as.character(externalFeatureId)])

    fulls[[i]] <- results(srs, result.type = "structureDb",
                          topFormula = 8, topStructure = 20,
                          return.type = "data.frame") |>
      mutate(across(any_of(sirius_id_cols), as.character),
             polarity = pol_lookup[as.character(xcms_fts)])

    if (has_cache) {
      write.csv(sums[[i]],  batch_sum_path,  row.names = FALSE)
      write.csv(fulls[[i]], batch_full_path, row.names = FALSE)
    }
    live_batches <- live_batches + 1L
  }

  non_empty <- function(x) Filter(\(df) nrow(df) > 0L, x)
  list(summary = bind_rows(non_empty(sums)),
       full    = bind_rows(non_empty(fulls)))
}

3. Extract + run SIRIUS per (mixture, lab)

Each (mixture, lab) is processed end-to-end before moving to the next: extract → run SIRIUS → write cache → drop spectra. Memory stays bounded to one lab’s spectra at a time, and a partial run resumes simply by re-rendering - cached pairs skip both extraction and SIRIUS.

# Skip 0-row tibbles inside each mixture's per-lab list before bind_rows: an
# empty cache from a (lab, mixture) with no MS2 features writes a header-only
# CSV, which read.csv parses with logical-typed columns. Combining those with
# real character/numeric columns would crash bind_rows.
agg <- function(results_list) {
  bind_rows(lapply(mixtures, function(m) {
    non_empty <- Filter(\(df) nrow(df) > 0L, results_list[[m]])
    if (length(non_empty) == 0L) return(NULL)

7.1 Build the flat peak table

EICs are extracted with a generous ±60 s symmetric window around the annotation RT. The wide window serves two purposes:

  1. Noise context. §7.4’s SNR-aware peak picker estimates baseline + noise from the EIC tails (|rt − apex| > 25 s); we need real tail region to compute MAD / IQR of intensity.
  2. Tolerance to cross-lab RT drift. ±60 s easily covers cembio ↔︎ icl gradient differences and early-RT instabilities, so the single pass below handles what the legacy two-pass approach handled.

The ±20 s pass-1 of the legacy pipeline (and its ±80 s pass-2 retry) have been removed — they were a workaround for peakBoundary()’s habit of locking onto the tallest local maximum even when no real peak is present (see §7.4 rationale).

The current build also addresses three MS2-hit-level issues identified in 9_per_compound_diagnostics.qmd:

  • All adducts get their own EIC. The legacy code did adduct = first(adduct) per (mixture, lab, polarity, compound), silently dropping every adduct beyond the first. We now group by adduct as well, so [M+H]+, [M+Na]+, [M+K]+, [M-H+formic]- etc. each emit their own row in the canonical CSV.

  • All features get a vote (no median collapse). The legacy code did rt_sec = median(rt_sec) per (mixture, lab, polarity, compound, adduct). For cembio’s block-MS2 mode, extract_all_ms2()’s rle(precursorMz) can shatter ONE chromatographic peak into many features when other inclusion-list precursors interleave — e.g. ritalinic acid in cembio mix 2.9 pos has 11 features spanning RT 198–258 s. The median (227 s) isn’t any of them. We now keep every feature as a candidate so the clustering step decides which RTs belong to one peak.

  • Cluster-based consensus RT combines cross-lab, cross-polarity, and cross-feature evidence to pick the “true” RT for every (mixture, compound). The algorithm:

    1. Pool all candidate RTs from every (lab, polarity, adduct, feature) tuple that SIRIUS annotated for this compound.
    2. Greedy 1-D clustering with a 30 s window (configurable via rt_cluster_window).
    3. Score each cluster as n_labs × 10 + n_polarities × 5 + sum_cosmic. Highest score wins.
    4. The winning cluster’s mean RT becomes the consensus RT used for every lab’s chromExtract of that compound, in both polarities.
    5. Labs whose own MS2 fell in the winning cluster get source = "own"; labs whose MS2 fell in a non-winning cluster (DDA hit the wrong peak) get source = "consensus" — their wrong RT no longer contaminates the propagation. Labs that never annotated this compound also get source = "consensus".

    This naturally fixes the cross-polarity disagreement, the cembio feature-shattering, and the wrong-peak DDA-trigger problem in one pass.

# --- RT clustering per (mixture, compound)
#
# For each (mixture, compound), pool ALL candidate RTs from every
# (lab, polarity, adduct, feature) tuple. Cluster them with a greedy
# 1-D agglomeration: sort RTs, walk the sorted list, start a new
# cluster whenever the gap between consecutive RTs exceeds
# `rt_cluster_window`. This naturally collapses cembio's 11
# ritalinic-acid pos features (5–6 s apart) into one cluster while
# separating the wrong-peak co-eluting contaminants (≥ 30 s away).
rt_cluster_window <- 30   # s

cluster_rts_greedy <- function(rt) {
  if (length(rt) == 0L) return(integer(0))
  ord <- order(rt); s <- rt[ord]
  out <- integer(length(rt))
  cl <- 1L; out[ord[1L]] <- cl
  if (length(s) >= 2L) for (i in 2:length(s)) {
    if (s[i] - s[i - 1L] > rt_cluster_window) cl <- cl + 1L
    out[ord[i]] <- cl
  }
  out
}

7.4 SNR-aware peak picking — rationale

The previous pipeline used Chromatograms::peakBoundary() to find the tallest local maximum in each EIC, then re-extracted on its boundary. For real peaks this works fine, but it has a critical failure mode: when the compound is below detection at a given lab, the EIC is mostly noise, and peakBoundary() happily returns the tallest noise spike as the apex. The two-pass widening retry doesn’t help — it just gives the noise more room to find a louder spike.

We saw this in raw-file diagnostics: hmgu’s NAPS standards are the strongest of any lab (geo-mean 3.5 × 10⁵ counts in pos), but ~70 % of its annotated spike-in rows reported maxIntensity in the 1-100 count range with apex_rt scattered across the full chromatogram. Those “measurements” were noise — the chromatographic peak simply wasn’t there at the dilution used. Sciex ESI source ionisation efficiency appears compound-specific; a panel tuned for Agilent Jet Stream falls under hmgu’s per-compound floor for a chunk of the panel.

The new picker applies three independent rejection guards. A row must pass all three to be reported as detection = "detected":

  1. SNR. apex_int − baseline must exceed snr_threshold × noise, where baseline = median of intensities at |rt − apex| > 25 s and noise = max(MAD, IQR/2, 1) of the low-intensity portion of the tail only (intensities below the tail’s 50 th percentile). The bottom-half restriction matters because at the 20 ppm m/z window used in §7.1, EICs frequently contain other chromatographic peaks in the tail region (isomers, late-eluting contaminants at the same m/z). Including those in the noise estimate makes “noise” the size of a peak, not the size of background, and crushes the SNR of real apex peaks (e.g. cembio L-phenylalanine got noise = 894 k, apex = 4.6 M → SNR ≈ 0; bottom-half tail gives noise ≈ 75 k, SNR ≈ 60). The IQR floor matters for sparse-positive data: MAD goes to 0 when the tail is mostly exact zeros, IQR catches the spread of low-count noise spikes that MAD misses.

  2. Shape. The apex’s immediate neighbours (apex index ±2, raw, not smoothed) must also be elevated. We require ≥ 2 of those 5 scans to be ≥ 30 % of apex_int. A single-scan spike with zero-neighbours = noise. A real chromatographic peak has a rising and falling shoulder spanning at least the next-scan ↔︎ apex pair.

  3. Absolute floor (only when noise estimate is at its hard floor). If both MAD and IQR are exactly 0, the tail region had no useful noise context (a “flat-zero EIC, except for one tiny spike at the prior”). We can’t trust SNR there; require apex_int ≥ 30 counts absolute. This kills the residual hmgu noise grabs that the SNR guard alone passes because (5.7 − 0) / (floor-noise=1) = 5.7.

Each row also gets a source-aware RT prior:

  • prior_kind = "own": the lab annotated this compound via SIRIUS and its COSMIC ≥ 0.1 and its own apex_rt agrees with the cross-lab consensus within 30 s. Search ±8 s around the lab’s own RT (tight — SIRIUS already told us where the peak is).
  • prior_kind = "consensus": everything else. Search ±15 s around the cross-lab median of “own” RTs (wider — accommodates inter-lab chromatographic drift). This includes “own” rows the demotion guards disagreed with (missing COSMIC, or RT > 30 s off consensus — a clue the SIRIUS hit was spurious).

Returns include the metrics (maxIntensity, areaUnderTic, xicFwhm), the apex_rt, and the diagnostic columns (snr, baseline, noise, n_scans_in_shape, prior_kind, detection).

7.6 Picker function and per-row extraction

extract_one <- function(rt, intensity, rt_prior,
                        apex_search_pad = 10,
                        tail_pad        = 25,
                        snr_threshold   = 5,
                        abs_min_counts  = 30) {
  ok <- !is.na(intensity)
  rt <- rt[ok]; intensity <- intensity[ok]
  if (length(rt) < 5L)
    return(list(apex_rt = NA_real_, maxIntensity = NA_real_,
                areaUnderTic = NA_real_, xicFwhm = NA_real_,
                snr = NA_real_, baseline = NA_real_, noise = NA_real_,
                n_scans_in_shape = 0L, detection = "no_data"))

  ## centred 3-point moving average (smoothed trace for valley/noise only)
  sm <- as.numeric(stats::filter(intensity, rep(1, 3) / 3, sides = 2))
  sm[is.na(sm)] <- intensity[is.na(sm)]

  ## (1) apex on RAW intensity inside prior ± apex_pad
  search_idx <- which(rt >= rt_prior - apex_search_pad &
                      rt <= rt_prior + apex_search_pad)
  if (!length(search_idx))
    return(list(apex_rt = NA_real_, maxIntensity = NA_real_,
                areaUnderTic = NA_real_, xicFwhm = NA_real_,
                snr = NA_real_, baseline = NA_real_, noise = NA_real_,
                n_scans_in_shape = 0L, detection = "no_data"))
  apex_local <- search_idx[which.max(intensity[search_idx])]
  apex_rt    <- rt[apex_local]
  apex_int   <- intensity[apex_local]

  ## (2) noise + baseline from tail (|rt - apex| > tail_pad)
  tail_idx <- which(abs(rt - apex_rt) > tail_pad)
  if (length(tail_idx) < 10L) {
    n <- length(rt); tail_idx <- c(seq_len(round(0.25 * n)),
                                    (n - round(0.25 * n) + 1L):n)
  }
  tail_int <- intensity[tail_idx]
  ## Use ONLY the low-intensity portion of the tail (bottom 50 %) for
  ## noise/baseline. Otherwise other chromatographic peaks at the same
  ## m/z window (isomers, late-eluting contaminants) get treated as
  ## "noise" and crush the SNR of real peaks. See §7.4 guard #1.
  tail_low <- tail_int[tail_int <= quantile(tail_int, 0.5, na.rm = TRUE)]
  baseline <- median(tail_low, na.rm = TRUE)
  noise_mad <- mad(tail_low, na.rm = TRUE)
  q15 <- as.numeric(quantile(tail_low, 0.15, na.rm = TRUE))
  q85 <- as.numeric(quantile(tail_low, 0.85, na.rm = TRUE))
  raw_noise <- max(noise_mad, (q85 - q15) / 2, na.rm = TRUE)
  noise_at_floor <- !is.finite(raw_noise) || raw_noise <= 0
  noise <- max(raw_noise, 1, na.rm = TRUE)
  snr   <- (apex_int - baseline) / noise

  ## (3) valleys for peak boundaries on the smoothed trace
  v_idx <- tryCatch(MsCoreUtils::valleys(sm), error = function(e) integer(0))
  if (length(v_idx) >= 2L) {
    left  <- suppressWarnings(max(v_idx[v_idx <= apex_local], na.rm = TRUE))
    right <- suppressWarnings(min(v_idx[v_idx >= apex_local], na.rm = TRUE))
    if (!is.finite(left))  left  <- 1L
    if (!is.finite(right)) right <- length(rt)
  } else {
    left <- apex_local; right <- apex_local
    while (left  > 1L         && sm[left  - 1L] > baseline + noise) left  <- left  - 1L
    while (right < length(rt) && sm[right + 1L] > baseline + noise) right <- right + 1L
  }
  pk_rt <- rt[left:right]; pk_int <- intensity[left:right]
  area  <- if (length(pk_rt) >= 2L)
             sum(diff(pk_rt) * (pk_int[-1] + pk_int[-length(pk_int)]) / 2)
           else NA_real_
  hm    <- (apex_int - baseline) / 2 + baseline
  above <- which(pk_int >= hm)
  fwhm  <- if (length(above) >= 2L) diff(range(pk_rt[above])) else NA_real_

  ## (4) shape sanity: at least 2 of 5 scans around apex >= 30% of apex
  nb_lo  <- max(1L, apex_local - 2L); nb_hi <- min(length(rt), apex_local + 2L)
  n_shape <- sum(intensity[nb_lo:nb_hi] >= 0.3 * apex_int, na.rm = TRUE)

  ## (5) three-guard detection classification
  detection <- if (apex_int <= baseline) "below_detection"
               else if (snr < snr_threshold) "below_detection"
               else if (n_shape < 2L) "below_detection"
               else if (noise_at_floor && apex_int < abs_min_counts) "below_detection"
               else "detected"

  list(apex_rt = apex_rt,
       maxIntensity = if (detection == "detected") apex_int else NA_real_,
       areaUnderTic = if (detection == "detected") area     else NA_real_,
       xicFwhm     = if (detection == "detected") fwhm     else NA_real_,
       snr = snr, baseline = baseline, noise = noise,
       n_scans_in_shape = n_shape, detection = detection)
}
